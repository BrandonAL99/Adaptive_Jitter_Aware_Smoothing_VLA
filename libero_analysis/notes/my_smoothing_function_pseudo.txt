def adaptive_jitter_smooth(action_chunk, tc_score):
    """
    Adaptive Jitter-Aware Action Smoothing (AJAS)
    
    Args:
        action_chunk: (50, action_dim) - the raw action chunk from VLA
        tc_score: float [0,1] - temporal consistency score (higher = smoother)
    
    Returns:
        smoothed_chunk: (50, action_dim) - smoothed actions
    """
    
    # 1. Map TC score to stride k (look-ahead distance)
    if tc_score < 0.3:
        k = 1       Very jittery → minimal smoothing
    elif tc_score < 0.6:
        k = 3      # Moderate jitter
    elif tc_score < 0.8:
        k = 5      # Low jitter
    else:
        k = 10     # Very smooth → aggressive interpolation
    
    # 2. Select keyframe indices
    keyframe_indices = [0, k, 2*k, 3*k, ...]  # Every k-th action
    
    # Always include final action (preserve endpoint)
    if keyframe_indices[-1] != 49:
        keyframe_indices.append(49)
    
    # Example with k=7, n=50:
    # keyframe_indices = [0, 7, 14, 21, 28, 35, 42, 49]
    
    # 3. Extract keyframe actions
    keyframes = action_chunk[keyframe_indices]
    
    # 4. Linear interpolation between keyframes
    smoothed_chunk = empty array (50, action_dim)
    
    for each consecutive pair of keyframes:
        start_idx = keyframe_indices[i]
        end_idx = keyframe_indices[i+1]
        
        # Interpolate between start and end
        for each action dimension:
            smoothed_chunk[start_idx:end_idx+1, dim] = 
                linspace(keyframes[i, dim], 
                        keyframes[i+1, dim], 
                        num_steps = end_idx - start_idx + 1)
    
    return smoothed_chunk

#         (much smoother than original!)
